private int valutaTabellone(CXBoard tabellone) {
    int punteggio = 0;

    // Valuta le righe
    for (int riga = 0; riga < M; riga++) {
        for (int colonna = 0; colonna < N - X + 1; colonna++) {
            punteggio += valutaSegmento(tabellone, riga, colonna, 0, 1);  // orizzontale
        }
    }

    // Valuta le colonne
    for (int colonna = 0; colonna < N; colonna++) {
        for (int riga = 0; riga < M - X + 1; riga++) {
            punteggio += valutaSegmento(tabellone, riga, colonna, 1, 0);  // verticale
        }
    }

    // Valuta le diagonali (da sinistra a destra)
    for (int riga = 0; riga < M - X + 1; riga++) {
        for (int colonna = 0; colonna < N - X + 1; colonna++) {
            punteggio += valutaSegmento(tabellone, riga, colonna, 1, 1);  // diagonale
        }
    }

    // Valuta le diagonali (da destra a sinistra)
    for (int riga = X - 1; riga < M; riga++) {
        for (int colonna = 0; colonna < N - X + 1; colonna++) {
            punteggio += valutaSegmento(tabellone, riga, colonna, -1, 1);  // diagonale
        }
    }

    return punteggio;
}

private int valutaSegmento(CXBoard tabellone, int rigaInizio, int colonnaInizio, int incrementoRiga, int incrementoColonna) {
    int punteggio = 0;

    // Controlla i seguenti X gettoni
    for (int k = 0; k < X; k++) {
        CXCellState cella = tabellone.cellState(rigaInizio + k*incrementoRiga, colonnaInizio + k*incrementoColonna);

        // Aggiungi al punteggio in base allo stato della cella
        if (cella == CXCellState.ME) {
            punteggio += 1;  // o qualunque sia la tua euristica
        } else if (cella == CXCellState.OPPONENT) {
            punteggio -= 1;  // o qualunque sia la tua euristica
        }
    }

    return punteggio;
}


@Override
public int selectColumn(CXBoard B) {
    // Prepara una lista per memorizzare i punteggi delle colonne
    int[] scores = new int[N];
    for (int col = 0; col < N; col++) {
        if (!B.fullColumn(col)) {
            // Simula la mossa nella colonna
            B.markColumn(col);
            // Valuta la mossa
            scores[col] = evaluateBoard(B);
            // Annulla la mossa simulata
            B.unmarkColumn();
        } else {
            // Se la colonna è piena, assegna un punteggio molto basso
            scores[col] = Integer.MIN_VALUE;
        }
    }

    // Scegli la colonna con il punteggio più alto
    int bestCol = 0;
    int bestScore = scores[0];
    for (int col = 1; col < N; col++) {
        if (scores[col] > bestScore) {
            bestScore = scores[col];
            bestCol = col;
        }
    }

    return bestCol;
}


"M - X + 1" e "N - X + 1" sono usati per determinare il limite superiore per i cicli di valutazione delle righe,
colonne e diagonali.

Ricorda che l'obiettivo del gioco Connect X è di allineare X gettoni in una fila. Quindi, se stai cercando
di valutare una possibile sequenza di X gettoni in una riga,
non ha senso iniziare a cercare in una colonna che sia più lontana da "N - X + 1",
perché non ci sarebbero abbastanza colonne rimanenti per formare una linea di X gettoni.

Lo stesso vale per la valutazione delle colonne e delle diagonali.
Non iniziare a cercare in una riga che sia più in basso di "M - X + 1",
perché non ci sarebbero abbastanza righe rimanenti sotto per formare una linea di X gettoni.

In sostanza, "M - X + 1" e "N - X + 1" ti assicurano di non cercare sequenze di X gettoni in posizioni
in cui non potrebbero esistere a causa delle dimensioni della griglia di gioco.


public int selectColumn(CXBoard tabellone) {
    int colonnaMigliore = -1;
    double punteggioMigliore = Double.NEGATIVE_INFINITY;

    long tempoLimite = System.currentTimeMillis() + timeout_in_secs * 1000;

    for (int profondita = 1; System.currentTimeMillis() < tempoLimite; profondita++) {
        for (int colonna : tabellone.getAvailableColumns()) {
            double punteggio = ricercaIterativeDeepening(tabellone.copy(), colonna, profondita, Double.NEGATIVE_INFINITY, Double.POSITIVE_INFINITY, tempoLimite);
            if (punteggio > punteggioMigliore) {
                punteggioMigliore = punteggio;
                colonnaMigliore = colonna;
            }
        }
    }

    return colonnaMigliore;
}

private double ricercaIterativeDeepening(CXBoard tabellone, int colonna, int profondita, double alfa, double beta, long tempoLimite) {
    if (System.currentTimeMillis() >= tempoLimite || profondita == 0 || tabellone.gameState() != CXGameState.OPEN) {
        return valutaTabellone(tabellone);
    }

    tabellone.markColumn(colonna);

    double punteggioMax = Double.NEGATIVE_INFINITY;
    for (int col : tabellone.getAvailableColumns()) {
        double punteggio = -ricercaIterativeDeepening(tabellone.copy(), col, profondita - 1, -beta, -alfa, tempoLimite);
        punteggioMax = Math.max(punteggioMax, punteggio);
        alfa = Math.max(alfa, punteggio);
        if (alfa >= beta) {
            break;
        }
    }

    return punteggioMax;
}

Certo, puoi utilizzare questi metodi per adattare l'implementazione di `selectColumn` e `ricercaIterativeDeepening`.
Ho considerato l'utilizzo di `markColumn`, `unmarkColumn`, `getAvailableColumns` e `gameState`. Ecco il codice modificato:

```java
public int selectColumn(CXBoard tabellone) {
    int colonnaMigliore = -1;
    double punteggioMigliore = Double.NEGATIVE_INFINITY;

    long tempoLimite = System.currentTimeMillis() + timeout_in_secs * 1000;

    for (int profondita = 1; System.currentTimeMillis() < tempoLimite; profondita++) {

            double punteggio = ricercaIterativeDeepening(tabellone, profondita, Double.NEGATIVE_INFINITY, Double.POSITIVE_INFINITY, tempoLimite);
            tabellone.unmarkColumn();
            if (punteggio > punteggioMigliore) {
                punteggioMigliore = punteggio;
                colonnaMigliore = colonna;
            }
        }
    }

    return colonnaMigliore;
}

private double ricercaIterativeDeepening(CXBoard tabellone, int profondita, double alfa, double beta, long tempoLimite) {
    if (System.currentTimeMillis() >= tempoLimite || profondita == 0 || tabellone.gameState() != CXGameState.OPEN) {
        return valutaTabellone(tabellone);
    }

    double punteggioMax = Double.NEGATIVE_INFINITY;
    for (int col : tabellone.getAvailableColumns()) {
        tabellone.markColumn(col);
        double punteggio = -ricercaIterativeDeepening(tabellone, profondita - 1, -beta, -alfa, tempoLimite);
        tabellone.unmarkColumn();
        punteggioMax = Math.max(punteggioMax, punteggio);
        alfa = Math.max(alfa, punteggio);
        if (alfa >= beta) {
            break;
        }
    }

    return punteggioMax;
}
```

Nota che ora, in entrambi i metodi, la tabella del gioco viene modificata e poi ripristinata utilizzando i metodi `markColumn` e `unmarkColumn`.
Questo è necessario per eseguire la simulazione di ogni mossa possibile e poi ripristinare lo stato del gioco alla sua condizione originale.
Inoltre, il metodo `gameState` viene utilizzato per verificare se la partita è ancora aperta prima di valutare il tabellone.

Naturalmente, l'implementazione esatta di `valutaTabellone` dipenderà da come scegli di valutare lo stato del gioco.
Di seguito, troverai un esempio generico che utilizza alcuni dei metodi disponibili.
In questo esempio, se il giocatore ha vinto, la funzione ritorna un valore positivo infinito.
Se l'avversario ha vinto, ritorna un valore negativo infinito. Se il gioco è ancora in corso,
il punteggio è calcolato come la differenza tra il numero di celle marcate dal giocatore
e il numero di celle marcate dall'avversario.

```java
private double valutaTabellone(CXBoard tabellone) {
    CXGameState statoGioco = tabellone.gameState();
    if (statoGioco == CXGameState.WIN1 && primoGiocatore) {
        return Double.POSITIVE_INFINITY;
    } else if (statoGioco == CXGameState.WIN2 && !primoGiocatore) {
        return Double.POSITIVE_INFINITY;
    } else if (statoGioco == CXGameState.WIN1 || statoGioco == CXGameState.WIN2) {
        return Double.NEGATIVE_INFINITY;
    } else {
        int gettoniGiocatore = 0;
        int gettoniAvversario = 0;
        CXCell[] celleMarcati = tabellone.getMarkedCells();
        for (CXCell cella : celleMarcati) {
            if (cella.player() == (primoGiocatore ? 1 : 2)) {
                gettoniGiocatore++;
            } else {
                gettoniAvversario++;
            }
        }
        return gettoniGiocatore - gettoniAvversario;
    }
}
```

In questo esempio, ho supposto che la variabile `primoGiocatore`
sia un campo della classe che indica se il giocatore di software è il primo a giocare.
Ho utilizzato il metodo `getMarkedCells` per contare il numero di gettoni marcati da ciascun giocatore.

Ricorda, questo è solo un esempio generico.
Come scegli di valutare il tabellone dovrebbe dipendere dalla tua specifica strategia di gioco.
Potresti voler considerare altri fattori, come la possibilità di creare o bloccare sequenze di X gettoni.
